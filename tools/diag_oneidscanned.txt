function onEidScanned(eid) {
  // Pas d'écran en attente d'un scan actuellement (ex. on est sur l'accueil,
  // ou sur un écran qui n'a pas de champ EID) : on ignore silencieusement,
  // exactement comme si le lecteur "tapait" un scan dans le vide.
  if (panneauContact) return;   // panneau de la bulle de contact ouvert : rien n'est écrit en aveugle dans l'écran caché dessous (cible gardée pour la fermeture)
  if (!activeScanTarget || !activeScanTarget.isConnected) { activeScanTarget = null; return; }
  activeScanTarget.value = eid;
  activeScanTarget.dispatchEvent(new Event('input', { bubbles: true }));
  activeScanTarget.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', keyCode: 13, bubbles: true }));
}

function btStopPolling() {
  if (btPollTimer) { clearInterval(btPollTimer); btPollTimer = null; }
}

  btStopPolling();
  const plugin = btPlugin();
  if (!plugin) return;
  btPollTimer = setInterval(async () => {
    try {
      const { available } = await plugin.available();
      if (available > 0) {
        const { data } = await plugin.read();
        const bytes = [];
        for (let i = 0; i < data.length; i++) bytes.push(data.charCodeAt(i) & 0xFF);
        const hex = bytes.map(b => b.toString(16).padStart(2, '0')).join(' ');
        const ascii = bytes.map(b => (b >= 32 && b <= 126) ? String.fromCharCode(b) : '.').join('');
        btDiag.frames.unshift({ ts: new Date().toLocaleTimeString('fr-FR'), hex, ascii, len: bytes.length });
        if (btDiag.frames.length > 200) btDiag.frames.length = 200; // borne la mémoire — outil de diagnostic, pas persisté
        if (currentView === 'diag-bluetooth') renderDiagBluetooth();

        // Reconstitution des trames complètes STX(0x02)...CRLF, indépendamment
        // du découp
